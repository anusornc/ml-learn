# กฎการเรียนรู้ของ Perceptron (Perceptron Learning Rule)
def fit(self, X, y):
    for epoch in range(self.n_epochs):
        errors = 0
        for xi, yi in zip(X, y):
            y_hat = self.predict(xi)   # (1) ทำนายคลาส 0 หรือ 1
            error = yi - y_hat         # (2) คำนวณข้อผิดพลาด
            if error != 0:
                for j in range(len(self.weights)):
                    self.weights[j] += self.learning_rate * error * xi[j] # (3) อัปเดต Weight
                self.bias += self.learning_rate * error                   # (4) อัปเดต Bias
